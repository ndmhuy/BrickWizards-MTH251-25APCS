\subsection{Problem 18. Improper integrals and comparison}
Consider the improper integral
\[J(p)=\displaystyle\int_{1}^{\infty}\frac{1}{x^{p}}dx, \quad p>0\]

\begin{enumerate}[label=(\alph*)]
    \item Evaluate $J(p)$ explicitly for all $p>1$, and show that it diverges for $0<p\le1$.
    \item Now consider the integral $\displaystyle\int_{1}^{\infty}\frac{\ln x}{x^{p}}dx$. Using comparison or limit comparison, determine for which values of $p$ this integral converges.
    \item For a specific value of $p$ where convergence holds (for example $p=3$), compute the integral $\displaystyle\int_{1}^{\infty}\frac{\ln x}{x^{3}}dx$ explicitly using integration by parts.
    \item Explain how these examples illustrate the idea that a slowly growing factor like $\ln x$ can still be dominated by a sufficiently fast decaying power $x^{-p}$.
\end{enumerate}

\subsection*{Strategy}
\begin{enumerate}[label=(\alph*)]
    \item We will evaluate the integral by separating the problem into two distinct cases based on the power rule for integration: the logarithmic case ($p=1$) and the power function case ($p \neq 1$). This will establish the standard convergence criteria for p-integrals.

    \item We will use the Limit Comparison Test. The key strategy is to compare this function with a standard p-integral $\frac{1}{x^q}$. Since $\ln x$ grows slower than any positive power of $x$, we aim to show that for certain $p$, the decay of $x^{-p}$ is sufficient to overcome the growth of $\ln x$.

    \item For the specific case $p=3$, we will compute the value of the improper integral by using Integration by Parts, choosing $u = \ln x$ to eliminate the logarithm, and $dv = x^{-3}dx$. We will evaluate the resulting antiderivative at the limits of integration.

    \item Finally, we will synthesize the results to explain the hierarchy of growth rates. The discussion will focus on how the "fast" polynomial decay of the term $x^{-p}$ (for $p>1$) dominates the "slow" logarithmic growth of $\ln x$, ensuring that the area under the curve remains finite despite the numerator approaching infinity.
\end{enumerate}

\subsection*{Solution}
\textbf{(a)}
For $p > 1$:
\begin{align*}
J(p) &= \lim_{t \to \infty} \int_{1}^{t} x^{-p} dx = \lim_{t \to \infty} \left[ \frac{x^{1-p}}{1-p} \right]_{1}^{t} \\
&= \frac{1}{1-p} \lim_{t \to \infty} \left( t^{1-p} - 1\right) = \frac{1}{1-p} (0 - 1) = \frac{1}{p-1}
\end{align*}
$\implies J(p)$ converges for $p > 1$.\\
For $p=1$:
    $$J(1) = \lim_{t \to \infty} \int_{1}^{t} \frac{1}{x} dx = \lim_{t \to \infty} \left[ \ln|x| \right]_{1}^{t} = \infty.$$
    $\implies J(p)$ diverges for $p=1$.\\
For $0 < p < 1$:
    $$J(p) = \frac{1}{1-p} \lim_{t \to \infty} \left( t^{1-p} - 1 \right) = \infty \quad \text{(since } 1-p > 0).$$
    $\implies J(p)$ diverges for $0 < p < 1$.\\
Hence, the integral $J(p)$ converges if and only if $p > 1$.


\textbf{(b)}
For $0<p\le1$, since $\frac{\ln x}{x^p}\ge\frac{1}{x^p}\quad \forall x\ge e$ and $\int_e^\infty\frac{1}{x^p}$ diverges, $\int_e^\infty\frac{\ln x}{x^p}$ also diverges.\\
For $p>1$, let $q$ a real number such that $1<q<p$. Let $\delta = p-q > 0$.\\
Let $f(x)=\frac{\ln x}{x^p}$ and $g(x) = \frac{1}{x^q}$, we have:
$$\lim_{x \to \infty}\frac{f(x)}{g(x)}=\lim_{x \to \infty} \frac{\ln x}{x^p}:\frac{1}{x^q} = \lim_{x \to \infty} \frac{\ln x}{x^{p-q}} = \lim_{x \to \infty} \frac{\ln x}{x^\delta}$$
Using L'Hôpital's Rule:
$$\lim_{x \to \infty} \frac{1/x}{\delta x^{\delta-1}} = \lim_{x \to \infty} \frac{1}{\delta x^\delta} = 0\quad\text{(since }\delta>0)$$
Since $\lim_{x \to \infty}\frac{f(x)}{g(x)} = 0$ and $\int_{1}^{\infty} g(x) dx$ converges (according to (a)), applying the Limit Comparison Test, we get: the integral $\int_{1}^{\infty} \frac{\ln x}{x^p} dx$ converges for $p > 1$.

\textbf{(c)}
Using $u = \ln x$ and $dv = \dfrac{1}{x^3} dx$:
\begin{align*}
    \int_{1}^{\infty} \frac{\ln x}{x^3} dx &= \lim_{t\to\infty}\int_1^t\frac{\ln x}{x^3}\\
    &=\lim_{t \to \infty}\left\{ \left[-\frac{\ln x}{2x^2}\right]_{1}^{t} - \int_1^t \left( -\frac{1}{2x^2} \frac{1}{x} dx \right) \right\} \\
    &= \lim_{t \to \infty} \left\{\left( -\frac{\ln t}{2t^2} + \frac{\ln 1}{2(1)^2} \right) + \frac{1}{2} \int_{1}^{t} \frac{1}{x^3} dx\right\}
\end{align*}
Applying L'Hôpital, we get: $\lim_{t \to \infty} \dfrac{\ln t}{2t^2} = \lim_{t \to \infty} \dfrac{1/t}{4t} = 0$\\ And according to (a): $\lim_{t\to\infty}\int_{1}^{t} \frac{1}{x^3} dx=\frac{1}{3-1}=\frac{1}{2}$. Therefore:
$$\int_{1}^{\infty} \frac{\ln x}{x^3} dx=\frac{1}{4}$$

\textbf{(d)} Explain how these examples illustrate the idea that a slowly growing factor like $\ln x$ can still be dominated by a sufficiently fast decaying power $x^{-p}$.

The integrals show that the convergence condition for $\int_{1}^{\infty} x^{-p} dx$ is $p > 1$. The factor $\ln x$, which grows slower than any positive power of $x$ does not change this convergence criterion. If $p$ is sufficiently large ($p > 1$), the polynomial decay of $x^{-p}$ is fast enough to dominate the slow logarithmic growth, ensuring the integral remains finite.
